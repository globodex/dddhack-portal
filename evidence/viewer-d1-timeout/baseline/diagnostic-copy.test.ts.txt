
function diagnostic() {
  const start = performance.now(); let previous = start;
  let rpc = 0, statements = 0, oldRpc = 0, oldStatements = 0;
  let previousCpu = process.cpuUsage();
  const phases: any[] = [], groups = new Map<string, any>();
  const stats = new Map<any, { raw: any, kind: string }>();
  const kind = (sql: string) => sql.includes("SELECT email FROM registrations WHERE email=? UNION SELECT email FROM luma_roster WHERE email=?") ? "registered-or-luma" : sql.trim() === "SELECT email FROM luma_roster WHERE email=?" ? "luma-only" : sql.trim().split(/\s+/, 2).join(" ");
  const record = async (k: string, count: number, work: () => Promise<any>) => {
    rpc++; statements += count; const t = performance.now();
    try { return await work(); } finally {
      const item = groups.get(k) || { rpc: 0, statements: 0, elapsedMs: 0 };
      item.rpc++; item.statements += count; item.elapsedMs += performance.now() - t; groups.set(k, item);
    }
  };
  function wrapStatement(raw: any, k: string): any {
    const wrapped = { bind(...args: any[]) { return wrapStatement(raw.bind(...args), k); }, all() { return record(k, 1, () => raw.all()); } };
    stats.set(wrapped, { raw, kind: k }); return wrapped;
  }
  return {
    wrap(db: any) { return { prepare(sql: string) { return wrapStatement(db.prepare(sql), kind(sql)); }, batch(items: any[]) { return record("BATCH", items.length, () => db.batch(items.map(item => stats.get(item)!.raw))); } }; },
    mark(name: string) { const current = performance.now(), cpu = process.cpuUsage(); const item = { phase: name, wallMs: current - previous, elapsedMs: current - start, rpc: rpc - oldRpc, statements: statements - oldStatements, nodeCpuMs: (cpu.user - previousCpu.user + cpu.system - previousCpu.system) / 1000 }; phases.push(item); console.log("D1_PHASE " + JSON.stringify(item)); previous = current; oldRpc = rpc; oldStatements = statements; previousCpu = cpu; },
    report() { return { phases, rpc, statements, wallMs: performance.now() - start, groups: Object.fromEntries(groups) }; }
  };
}
import { attend } from "/workspace/ddd-viewer-d1-baseline/tests/helpers";
import { test } from "node:test";
import assert from "node:assert/strict";
import { readFileSync, readdirSync } from "node:fs";
import { Miniflare } from "miniflare";
import { seedLocal, fixtures, LOCAL_CREDIT_KEY } from "/workspace/ddd-viewer-d1-baseline/scripts/fixtures";
import { PortalService } from "/workspace/ddd-viewer-d1-baseline/lib/service";
import { batch, rows, type Database } from "/workspace/ddd-viewer-d1-baseline/lib/db";
import { rehearsalPolicy, submission } from "/workspace/ddd-viewer-d1-baseline/tests/helpers";
test(
  "actual local D1 binding: schema, trigger rollback, batch atomicity and concurrent credit/submission commands",
  { timeout: 120000 },
  async () => {
    const diag = diagnostic();
    const mf = new Miniflare({
      modules: true,
      script:
        'export default {fetch(){return new Response("local D1 rehearsal")}}',
      compatibilityDate: "2026-07-01",
      cf: false,
      d1Databases: { DB: "portal-test" },
    });
    try {
      const native = await mf.getD1Database("DB");
      diag.mark("miniflare-binding");
      const db: Database = diag.wrap(native);
      const directory = "/workspace/ddd-viewer-d1-baseline/db/migrations";
      const sql = readdirSync(directory)
        .filter((n) => n.endsWith(".sql"))
        .sort()
        .map((n) => readFileSync(directory + "/" + n, "utf8"))
        .join("\n");
      const migrationFiles = readdirSync(directory).filter(n => n.endsWith(".sql")).sort();
      const migrationEnds = migrationFiles.map(n => [n, readFileSync(directory + "/" + n, "utf8").split("\n").length] as const);
      let migrationIndex = 0, migrationLine = 0;
      let buffer = "";
      for (const line of sql.split("\n")) {
        if (migrationIndex < migrationEnds.length && migrationLine++ === migrationEnds[migrationIndex][1]) {
          diag.mark("migration:" + migrationEnds[migrationIndex][0]);
          migrationIndex++; migrationLine = 1;
        }
        buffer += line + "\n";
        if (
          line.trim().endsWith(";") &&
          (!buffer.trim().startsWith("CREATE TRIGGER") ||
            line.trim() === "END;" ||
            (line.trim().startsWith("BEGIN ") && /\bEND;$/.test(line.trim())) ||
            line.trim().startsWith("CREATE TRIGGER"))
        ) {
          await db.prepare(buffer).all();
          buffer = "";
        }
      }
      assert.equal(buffer.trim(), "");
      diag.mark("migration:" + migrationEnds.at(-1)![0]);
      await seedLocal(db);
      diag.mark("seed");
      await batch(db, [
        [
          "UPDATE event_settings SET policy=?,active=0 WHERE id=1",
          JSON.stringify({ ...rehearsalPolicy, operationScopes: [] }),
        ],
      ]);
      let now = "2026-10-22T08:00:00.000Z";
      const admin = new PortalService(
        db,
        fixtures.admin,
        LOCAL_CREDIT_KEY,
        () => now,
      );
      const owner = new PortalService(
        db,
        fixtures.owner,
        LOCAL_CREDIT_KEY,
        () => now,
      );
      // No manual activation is required for existing actual policy data.
      assert.equal((await owner.settings()).active, 0);
      assert.equal(
        ((await owner.view("/admin/settings")).readiness as any).checkin
          .allowed,
        true,
      );
      diag.mark("settings-and-owner-view");
      const roster =
        "email\n" +
        Array.from({ length: 200 }, (_, i) => `bulk${i}@portal.test`).join(
          "\n",
        );
      const preview = await admin.command("preview-import", {
        kind: "roster",
        csv: roster,
      });
      diag.mark("roster-preview-200");
      await admin.command("commit-import", {
        kind: "roster",
        csv: roster,
        digest: preview.digest,
      });
      diag.mark("roster-commit-200");
      assert.equal(
        (
          await rows(
            db,
            "SELECT email FROM registrations WHERE email LIKE 'bulk%@portal.test'",
          )
        ).length,
        200,
      );
      diag.mark("roster-count-assertion");
      await Promise.all(
        Array.from({ length: 12 }, () =>
          admin.command("check-in", {
            userId: "fake_alice",
            reason: "Local D1 repeated check-in",
          }),
        ),
      );
      diag.mark("check-in-12-concurrent");
      assert.equal(
        (await rows(db, "SELECT * FROM credit_allocations")).length,
        1,
      );
      diag.mark("allocation-count-assertion");
      await assert.rejects(
        batch(db, [
          [
            "INSERT INTO teams VALUES(?,?,?,?,?)",
            "bad-team",
            "Should roll back",
            "fake_alice",
            "",
            "2026-10-22T08:00:00.000Z",
          ],
          [
            "INSERT INTO memberships VALUES(?,?,?)",
            "fake_alice",
            "bad-team",
            "2026-10-22T08:00:00.000Z",
          ],
        ]),
      );
      assert.equal(
        (await rows(db, "SELECT * FROM teams WHERE id='bad-team'")).length,
        0,
      );
      diag.mark("batch-rollback-and-assertion");
      await batch(db, [
        [
          "UPDATE event_settings SET policy=?,active=0 WHERE id=1",
          JSON.stringify(rehearsalPolicy),
        ],
      ]);
      now = "2026-11-01T06:00:00.000Z";
      const leader = new PortalService(
        db,
        fixtures.alice,
        LOCAL_CREDIT_KEY,
        () => now,
      );
      diag.mark("submission-setup");
      const outcomes = await Promise.allSettled(
        Array.from({ length: 10 }, (_, i) =>
          leader.command("submit", {
            ...submission,
            projectName: "Attempt " + i,
          }),
        ),
      );
      diag.mark("submit-10-concurrent");
      assert.equal(outcomes.filter((o) => o.status === "fulfilled").length, 3);
      assert.equal(
        (await rows(db, "SELECT * FROM submission_versions")).length,
        3,
      );
      assert.equal((await rows(db, "SELECT * FROM astra_jobs")).length, 3);
    } finally {
      diag.mark("final-count-assertions");
      await mf.dispose();
      diag.mark("dispose");
      console.log("D1_DIAGNOSTIC " + JSON.stringify(diag.report()));
    }
  },
);
